# --- Site forms (/contact-us and the sitewide "Get a Quote" panel) ---------
# Every form POSTs to app/api/enquiry/route.ts, which emails the lead via SMTP.

# Gmail SMTP. SMTP_PASS is a Google App Password (Google account -> Security
# -> 2-Step Verification -> App passwords), NOT the normal Gmail password.
SMTP_HOST=smtp.gmail.com
SMTP_PORT=587
# "tls" = STARTTLS on 587, "ssl" = implicit TLS on 465.
SMTP_SECURE=tls
SMTP_USER=your-account@gmail.com
SMTP_PASS=your-16-char-app-password

# Optional. Gmail always sends "From" the authenticated account, so MAIL_FROM
# defaults to SMTP_USER; only the display name is really ours to choose.
MAIL_FROM_NAME=Secure House Website

# Optional - these are the defaults if unset. The only visible recipient(s),
# comma-separated.
MAIL_TO=info@secure-house.co.uk
# Optional - defaults to these if unset. Hidden copies, comma-separated.
MAIL_BCC=pallavi.p@brainminetech.com,priyanka.s@brainminetech.com,gaurav.b@brainminetech.com
