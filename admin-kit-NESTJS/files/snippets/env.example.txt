# Copy to .env.local for local development.
# In production (Hostinger, Vercel, Netlify...) set the SAME names in the host's
# "Environment variables" screen. Never commit real values. Never prefix these
# with NEXT_PUBLIC_ (that would expose them to the browser).

# ---------------------------------------------------------------------------
# Pick ONE of the three options below.
# ---------------------------------------------------------------------------

# --- Option A: Hostinger MySQL (site running ON Hostinger) -----------------
DB_TYPE=mysql
DB_HOST=localhost
DB_PORT=3306
DB_NAME=u123456789_dbname
DB_USER=u123456789_user
DB_PASSWORD=your-database-user-password

# --- Option B: Hostinger MySQL from your own computer or another host ------
# Turn on hPanel -> Databases -> Remote MySQL, add your IP, and use the
# hostname shown there instead of localhost:
# DB_TYPE=mysql
# DB_HOST=srv123.hstgr.io
# DB_PORT=3306
# DB_NAME=u123456789_dbname
# DB_USER=u123456789_user
# DB_PASSWORD=your-database-user-password

# --- Option C: Supabase / any PostgreSQL (one connection string) -----------
# Supabase: Project -> Connect -> "Transaction pooler" (port 6543) is best for
# Vercel/serverless; "Session pooler" (port 5432) also works. Put your database
# password in the string. URL-encode special characters (@ becomes %40).
# DB_TYPE=postgres
# DATABASE_URL=postgresql://postgres.xxxxxxxx:YOUR-PASSWORD@aws-0-region.pooler.supabase.com:6543/postgres

# Optional
# DB_SSL=true          # force SSL on/off (Postgres remote hosts default to on)
# DB_POOL_SIZE=5       # keep small on shared hosting
